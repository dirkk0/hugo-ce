{{- /* Image render hook. Content authors reference images loosely (../img/x.png, ../../img/x.png, etc.) which only resolve at one page depth. Since the news list inlines each post's full .Content, the same relative path breaks there. Fix: rewrite every local image to a baseURL-absolute /img/ URL so it resolves identically on list and detail pages regardless of depth. External URLs (http/https/protocol-relative) pass through untouched. The trailing #small / #verysmall fragment is preserved — CSS sizes on it via img[src$="#small"]. */ -}} {{- $dest := .Destination -}} {{- if or (hasPrefix $dest "http://") (hasPrefix $dest "https://") (hasPrefix $dest "//") -}} {{- else -}} {{- $u := urls.Parse $dest -}} {{- $base := path.Base $u.Path -}} {{- $src := printf "img/%s" $base | relURL -}} {{- with $u.Fragment }}{{ $src = printf "%s#%s" $src . }}{{ end -}} {{- /* images.Config errors on non-raster formats (e.g. SVG); skip dimensions for those rather than break the build */ -}} {{- $dims := "" -}} {{- if not (eq (lower (path.Ext $base)) ".svg") -}} {{- with (images.Config (printf "static/img/%s" $base)) -}} {{- $dims = printf ` width="%d" height="%d"` .Width .Height -}} {{- end -}} {{- end -}} {{- end -}}
